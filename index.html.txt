  <script>
    // --- PASSCODE UNLOCK ---
    function checkPasscode() {
      const input = document.getElementById('passcode').value;
      if (input === '1008') { // Passcode: 10th August
        document.getElementById('lock-screen').style.display = 'none';
        document.getElementById('main-content').style.display = 'block';
        initCarrom();
      } else {
        document.getElementById('error-msg').style.display = 'block';
      }
    }

    // --- GAME 1: CARROM ---
    let canvas, ctx;
    let striker = { x: 150, y: 170, radius: 10 };
    let queen = { x: 150, y: 80, radius: 8 };
    let pocket = { x: 270, y: 30, radius: 18 };
    let shotTaken = false;

    function initCarrom() {
      canvas = document.getElementById('carromCanvas');
      ctx = canvas.getContext('2d');
      drawCarrom();
      canvas.addEventListener('click', fireStriker);
    }

    function drawCarrom() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      // Pocket
      ctx.beginPath();
      ctx.arc(pocket.x, pocket.y, pocket.radius, 0, Math.PI * 2);
      ctx.fillStyle = '#333';
      ctx.fill();

      // Queen (Red)
      ctx.beginPath();
      ctx.arc(queen.x, queen.y, queen.radius, 0, Math.PI * 2);
      ctx.fillStyle = '#c9184a';
      ctx.fill();

      // Striker (White)
      ctx.beginPath();
      ctx.arc(striker.x, striker.y, striker.radius, 0, Math.PI * 2);
      ctx.fillStyle = '#ffffff';
      ctx.strokeStyle = '#000';
      ctx.lineWidth = 2;
      ctx.fill();
      ctx.stroke();
    }

    function fireStriker() {
      if (shotTaken) return;
      shotTaken = true;
      
      let anim = setInterval(() => {
        striker.y -= 5;
        if (striker.y <= queen.y + 10) {
          queen.x += 4;
          queen.y -= 2;
        }

        drawCarrom();

        if (queen.x >= pocket.x - 10 && queen.y <= pocket.y + 10) {
          clearInterval(anim);
          document.getElementById('carrom-result').innerText = "QUEEN POCKETED! You hit the bullseye in my heart! 🎯❤️";
          document.getElementById('carrom-result').style.color = "#c9184a";
        }
      }, 20);
    }

    function resetCarrom() {
      striker = { x: 150, y: 170, radius: 10 };
      queen = { x: 150, y: 80, radius: 8 };
      shotTaken = false;
      document.getElementById('carrom-result').innerText = "";
      drawCarrom();
    }

    // --- GAME 2: QUIZ ---
    function answerQuiz(isCorrect) {
      const result = document.getElementById('quiz-result');
      if (isCorrect) {
        result.innerText = "Correct! You remember every detail ❤️";
        result.style.color = "green";
      } else {
        result.innerText = "Nice try! Try again 😜";
        result.style.color = "red";
      }
    }

    // --- GAME 3: LUDO SIX ---
    let rollCount = 0;
    function rollDice() {
      rollCount++;
      const diceEl = document.getElementById('dice');
      const resultEl = document.getElementById('ludo-result');
      
      let roll = (rollCount >= 3) ? 6 : Math.floor(Math.random() * 5) + 1;
      diceEl.innerText = roll;

      if (roll === 6) {
        resultEl.innerText = "YOU ROLLED A 6! 🎉 You're always my lucky number!";
        resultEl.style.color = "green";
      } else {
        resultEl.innerText = `Rolled a ${roll}! Roll again for a 6! 🎲`;
        resultEl.style.color = "orange";
      }
    }
  </script>
